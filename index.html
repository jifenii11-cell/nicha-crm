/** สมัครสมาชิกและอัปเดตข้อมูลสมาชิกเดิม */ function registerMember(p) { checkHeaders(); if (!p.lineUserId) throw new Error('ไม่พบ LINE User ID'); if (!p.name) throw new Error('กรุณาระบุชื่อสมาชิก'); const phone = String(p.phone || '').replace(/\D/g, ''); if (!/^0[0-9]{9}$/.test(phone)) { throw new Error('กรุณากรอกเบอร์โทรศัพท์ 10 หลัก'); } const existing = findMemberRow(p.lineUserId); const sheet = getSheet('Members'); if (existing) { // อัปเดตข้อมูลสมาชิกเดิม sheet.getRange(existing._row, 2).setValue(p.name); sheet.getRange(existing._row, 3).setValue(phone); return { success: true, message: 'อัปเดตข้อมูลสมาชิกสำเร็จ', member: getMember(p.lineUserId).member }; } sheet.appendRow([ p.lineUserId, p.name, phone, new Date(), 0, 'Bronze' ]); return { success: true, message: 'สมัครสมาชิกสำเร็จ', member: getMember(p.lineUserId).member }; }
